  \begin{tabularx}{\linewidth}{@{}lX@{}}

  \vspace{0.4cm}

  \begin{itemize}
    \item 4 architectures valid\'ees / 70 questions
    \item 3 LLMs : GPT-3.5, Mistral~7B, Llama3
    \item 4 mod\`eles d'embeddings \'evalu\'es
    \item Prototype hybride temps r\'eel
  \end{itemize}
}
